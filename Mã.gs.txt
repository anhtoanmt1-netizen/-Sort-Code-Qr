/**
 * Hàm lấy dữ liệu từ sheet "DATA ALL" làm database động trên client
 * Link Sheet database: https://docs.google.com/spreadsheets/d/1KvkbTXTHlYc6ihgignux527_bg3BzIEKk5RHtd9Rnmw/edit#gid=746223085
 */
function getRawDataFromSheet() {
  try {
    const spreadsheetId = "1KvkbTXTHlYc6ihgignux527_bg3BzIEKk5RHtd9Rnmw";
    const ss = SpreadsheetApp.openById(spreadsheetId);
    
    // Lấy chính xác Sheet có tên "DATA ALL" thay vì vị trí [0] như phiên bản cũ
    const sheet = ss.getSheetByName("DATA ALL");
    
    if (!sheet) {
      Logger.log("Lỗi: Không tìm thấy sheet 'DATA ALL' trong file bảng tính.");
      return [];
    }
    
    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
    
    if (lastRow < 2) {
      return [];
    }
    
    const range = sheet.getRange(1, 1, lastRow, lastCol);
    const values = range.getValues();
    
    // --- XỬ LÝ DỮ LIỆU LOGIC MỚI: CHUẨN HÓA DATA VÀ LỌC LỖI ---
    // Giữ tiêu đề hàng thứ nhất
    const headerRow = values[0];
    const cleanedValues = [headerRow];
    
    // Tập hợp để lưu trữ dạng khóa duy nhất sau khi ép Chữ Hoa để loại bỏ các ký tự trùng lặp chữ hoa/chữ thường
    const uniqueKeys = new Set();
    
    // Hàm phụ trợ kiểm tra giá trị lỗi của Google Sheets
    const isErrorVal = function(val) {
      if (val === null || val === undefined) return true;
      const str = val.toString().trim();
      if (str.startsWith("#")) return true;
      const errors = ["#n/a", "#value!", "#ref!", "#div/0!", "#num!", "#name?", "#null!"];
      return errors.includes(str.toLowerCase());
    };

    // Hàm chuyển đổi sang Proper Case (Ví dụ: "Tỉnh Gia Lai")
    const toProperCase = function(str) {
      if (!str) return "";
      return str.toLowerCase().split(' ').map(function(word) {
        return word.charAt(0).toUpperCase() + word.slice(1);
      }).join(' ');
    };

    // Hàm loại bỏ tiền tố Tỉnh/Thành phố và khoảng trắng để tìm các giá trị Province tương đương nhau
    const cleanProvinceForMapping = function(str) {
      if (!str) return "";
      let cleaned = str.toLowerCase().trim();
      cleaned = cleaned.replace(/^(tỉnh|thành phố|thanh pho)\s+/i, "");
      return cleaned.replace(/\s+/g, "");
    };
    
    // --- BƯỚC 1: XÁC ĐỊNH BẢN ĐỒ CHUẨN HÓA CHO CÁC TỈNH (PROVINCE) ---
    // Duyệt trước dữ liệu để tìm ra tên đầy đủ ưu tiên nhất cho mỗi nhóm tỉnh tương đương
    const provinceMapping = {}; // Key: chuỗi rút gọn không khoảng trắng. Value: Chuỗi tên đầy đủ gốc tốt nhất tìm thấy
    
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      if (!row || row.length === 0) continue;
      let colC = row[2] !== undefined ? row[2].toString().trim() : "";
      
      if (isErrorVal(colC) || colC === "") continue;
      
      const cleanKey = cleanProvinceForMapping(colC);
      const currentBest = provinceMapping[cleanKey];
      
      if (!currentBest) {
        provinceMapping[cleanKey] = colC;
      } else {
        // Ưu tiên giữ lại tên đầy đủ nhất (chứa chữ "Tỉnh" hoặc "Thành phố") thay vì tên rút gọn
        const curHasPrefix = /^(tỉnh|thành phố|thanh pho)/i.test(colC);
        const bestHasPrefix = /^(tỉnh|thành phố|thanh pho)/i.test(currentBest);
        if (curHasPrefix && !bestHasPrefix) {
          provinceMapping[cleanKey] = colC;
        } else if (colC.length > currentBest.length && (curHasPrefix === bestHasPrefix)) {
          provinceMapping[cleanKey] = colC;
        }
      }
    }

    // --- BƯỚC 2: CHUẨN HÓA CHI TIẾT TỪNG DÒNG ---
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      if (!row || row.length === 0) continue;
      
      let ward = row[0] !== undefined ? row[0].toString().trim() : "";
      let colB = row[1] !== undefined ? row[1].toString().trim() : "";
      let colC = row[2] !== undefined ? row[2].toString().trim() : "";
      let sortCode = row[3] !== undefined ? row[3].toString().trim() : "";
      let hubName = row[4] !== undefined ? row[4].toString().trim() : "";
      
      // BỎ QUA nếu dòng trống hoặc chứa bất kỳ giá trị lỗi nào
      if (isErrorVal(ward) || isErrorVal(colB) || isErrorVal(colC) || isErrorVal(sortCode) || isErrorVal(hubName)) {
        continue;
      }
      
      if (ward === "" && colB === "" && colC === "" && sortCode === "" && hubName === "") {
        continue;
      }
      
      // Gán Province theo tên đã chuẩn hóa ở Bước 1
      const provinceCleanKey = cleanProvinceForMapping(colC);
      if (provinceMapping[provinceCleanKey]) {
        colC = provinceMapping[provinceCleanKey];
      }
      
      // Chuyển Province thành Proper Case
      colC = toProperCase(colC);
      
      // Ép các trường khác thành Chữ hoa (Trừ HUB NAME)
      ward = ward.toUpperCase();
      colB = colB.toUpperCase();
      colC = colC.toUpperCase();
      sortCode = sortCode.toUpperCase();
      hubName = hubName.trim(); // Giữ nguyên chữ Hoa/thường nguyên bản cho HUB NAME
      
      // Tạo khóa duy nhất để lọc trùng lặp dữ liệu dư thừa
      const uniqueStringKey = ward + "||" + colB + "||" + colC + "||" + sortCode + "||" + hubName;
      
      if (!uniqueKeys.has(uniqueStringKey)) {
        uniqueKeys.add(uniqueStringKey);
        cleanedValues.push([ward, colB, colC, sortCode, hubName]);
      }
    }
    
    return cleanedValues;
    
  } catch (err) {
    Logger.log("Lỗi trong getRawDataFromSheet: " + err.message);
    return [];
  }
}

/**
 * KIẾN TRÚC LOGGING SYSTEM REQUIREMENTS:
 * Ghi lại log hoạt động của người dùng trong phiên làm việc hiện hành vào Google Sheet lưu trữ.
 * Sheet đích: Sheet tên "Save" (hoặc tên bất kỳ được định dạng lại)
 * File đích: https://docs.google.com/spreadsheets/d/1KvkbTXTHlYc6ihgignux527_bg3BzIEKk5RHtd9Rnmw/edit?gid=511377991#gid=511377991
 */
function saveArchitectLogSystem(logPayload) {
  try {
    const spreadsheetId = "1KvkbTXTHlYc6ihgignux527_bg3BzIEKk5RHtd9Rnmw";
    const ss = SpreadsheetApp.openById(spreadsheetId);
    
    // Ghi dữ liệu trực tiếp vào sheet mang tên "Save"
    let sheet = ss.getSheetByName("Save");
    if (!sheet) {
      // Trường hợp sheet bị thay đổi tên từ phía Google Sheets, hệ thống sẽ tự tìm Sheet có id là 511377991
      const sheets = ss.getSheets();
      for (let i = 0; i < sheets.length; i++) {
        if (sheets[i].getSheetId() === 511377991) {
          sheet = sheets[i];
          break;
        }
      }
      
      // Nếu vẫn không thể tìm thấy bằng GID, tự động chọn sheet đầu tiên để không làm đứt gãy luồng hệ thống
      if (!sheet) {
        sheet = ss.getSheets()[0];
      }
    }
    
    const userSession = logPayload.userSession;
    const trackingModification = logPayload.trackingModification;
    
    const gmail = userSession.gmail;
    const sessionStart = userSession.sessionStart;
    const sessionEnd = userSession.sessionEnd;
    const totalDuration = userSession.totalDuration;
    const successCount = trackingModification.successCount;
    const details = trackingModification.details; // Mảng các chi tiết sửa đổi
    
    // Chuẩn bị dữ liệu để append vào sheet hàng loạt
    const rowsToAppend = [];
    
    // Cấu trúc cột đích: 
    // [Thời gian Bắt đầu truy cập, Thời gian Kết thúc truy cập, Tổng thời gian truy cập Wed, Tài khoản gmail truy cập, Số đơn đổi Sort Code, Mã Vận Đơn, Hub Name, Sort Code New]
    if (details && details.length > 0) {
      details.forEach(function(detail) {
        rowsToAppend.push([
          sessionStart,                 // Thời gian Bắt đầu truy cập
          sessionEnd,                   // Thời gian Kết thúc truy cập
          totalDuration,                // Tổng thời gian truy cập Wed
          gmail,                        // Tài khoản gmail truy cập
          successCount,                 // Số đơn đổi Sort Code
          detail.trackingNumber,        // Mã Vận Đơn
          detail.destinationHubName,    // Hub Name
          detail.destinationSortCode    // Sort Code New
        ]);
      });
    } else {
      // Trường hợp dự phòng nếu không có chi tiết cụ thể
      rowsToAppend.push([
        sessionStart,
        sessionEnd,
        totalDuration,
        gmail,
        successCount,
        "",
        "",
        ""
      ]);
    }
    
    // Thực thi ghi dữ liệu hàng loạt xuống Sheet Save
    if (rowsToAppend.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      const numRows = rowsToAppend.length;
      const numCols = rowsToAppend[0].length;
      sheet.getRange(startRow, 1, numRows, numCols).setValues(rowsToAppend);
    }
    
    return { success: true, count: rowsToAppend.length };
    
  } catch(err) {
    Logger.log("Lỗi trong saveArchitectLogSystem: " + err.message);
    return { success: false, error: err.message };
  }
}
